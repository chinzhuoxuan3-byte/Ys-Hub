local animEnabled = true
local animSpeed = 1

local rainbowSeq
do
	local keys = {}
	for i = 0, 6 do
		table.insert(keys, ColorSequenceKeypoint.new(i / 6, Color3.fromHSV(math.min(i / 6, 0.9999), 1, 1)))
	end
	rainbowSeq = ColorSequence.new(keys)
end

local function create(class, props, parent)
	local obj = Instance.new(class)
	for k, v in props do
		obj[k] = v
	end
	obj.Parent = parent
	return obj
end

local function corner(obj, radius)
	return create("UICorner", {CornerRadius = UDim.new(0, radius or 6)}, obj)
end

local function padding(obj, l, r, t, b)
	return create("UIPadding", {
		PaddingLeft = UDim.new(0, l or 0),
		PaddingRight = UDim.new(0, r or 0),
		PaddingTop = UDim.new(0, t or 0),
		PaddingBottom = UDim.new(0, b or 0)
	}, obj)
end

local function tween(obj, goal, time)
	TweenService:Create(obj, TweenInfo.new((time or 0.18) / animSpeed, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), goal):Play()
end

local function apply(obj, goal, animate)
	if animate and animEnabled then
		tween(obj, goal, 0.18)
	else
		for k, v in goal do
			obj[k] = v
		end
	end
end

local function randomName()
	local chars = {}
	for i = 1, math.random(12, 24) do
		chars[i] = string.char(math.random(97, 122))
	end
	return table.concat(chars)
end

local function isPointer(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function sheen(c, g, pos)
	pos = pos or 0.46
	return ColorSequence.new({
		ColorSequenceKeypoint.new(0, c:Lerp(W, 0.26 * g)),
		ColorSequenceKeypoint.new(pos, c:Lerp(W, 0.05 * g)),
		ColorSequenceKeypoint.new(pos + 0.04, c:Lerp(B, 0.12 * g)),
		ColorSequenceKeypoint.new(1, c:Lerp(B, 0.26 * g))
	})
end

local function hexColor(hex, fallback)
	local ok, c = pcall(Color3.fromHex, hex)
	return ok and c or fallback
end

local function gradSeq(c, th)
	local g = th.Gloss / 100
	local mode = th.GradMode
	if mode == "双色渐变" then
		return ColorSequence.new({
			ColorSequenceKeypoint.new(0, hexColor(th.GradTopHex, c)),
			ColorSequenceKeypoint.new(1, hexColor(th.GradBottomHex, c))
		})
	elseif mode == "金属拉丝" then
		local pts = {}
		for i = 0, 8 do
			local k = (i % 2 == 0) and c:Lerp(W, 0.2 * g) or c:Lerp(B, 0.1 * g)
			table.insert(pts, ColorSequenceKeypoint.new(i / 8, k))
		end
		return ColorSequence.new(pts)
	elseif mode == "玻璃" then
		return ColorSequence.new({
			ColorSequenceKeypoint.new(0, c:Lerp(W, 0.42 * g)),
			ColorSequenceKeypoint.new(0.14, c:Lerp(W, 0.08 * g)),
			ColorSequenceKeypoint.new(0.55, c),
			ColorSequenceKeypoint.new(1, c:Lerp(B, 0.2 * g))
		})
	end
	return sheen(c, g, th.GradPos / 100)
end

local function textShadow(label, th)
	local s = th.TextShadow
	if th.Perf or s == "关闭" then
		label.TextStrokeTransparency = 1
	else
		label.TextStrokeColor3 = B
		label.TextStrokeTransparency = s == "重" and 0.45 or 0.75
	end
end

local dividerNone = NumberSequence.new({NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(1, 0)})
local dividerFade = NumberSequence.new({NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(1, 1)})
local dividerDots
do
	local pts = {}
	for i = 0, 4 do
		local c = i * 0.2
		table.insert(pts, NumberSequenceKeypoint.new(c, 0))
		table.insert(pts, NumberSequenceKeypoint.new(c + 0.12, 0))
		table.insert(pts, NumberSequenceKeypoint.new(c + 0.121, 1))
		table.insert(pts, NumberSequenceKeypoint.new(i == 4 and 1 or c + 0.199, 1))
	end
	dividerDots = NumberSequence.new(pts)
end

local function rimSeq(p)
	return ColorSequence.new({
		ColorSequenceKeypoint.new(0, p.Rim1),
		ColorSequenceKeypoint.new(0.5, p.Rim2),
		ColorSequenceKeypoint.new(1, p.Rim1)
	})
end

local function borderSeq(th, p)
	local mode = th.FlowMode
	if mode == "彩虹" then
		return rainbowSeq
	elseif mode == "强调色" then
		return ColorSequence.new({
			ColorSequenceKeypoint.new(0, p.Accent),
			ColorSequenceKeypoint.new(0.5, p.Rim2),
			ColorSequenceKeypoint.new(1, p.Accent)
		})
	end
	return rimSeq(p)
end

local function makeGlass(obj, radius)
	obj.BackgroundColor3 = W
	local c = corner(obj, radius)
	local grad = create("UIGradient", {Rotation = 90}, obj)
	local stroke = create("UIStroke", {Thickness = 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, obj)
	local sg = create("UIGradient", {Rotation = 45}, stroke)
	local tracked = false
	local function style(window, p, base, r)
		local th = window.Theme
		local perf = th.Perf
		obj.BackgroundColor3 = perf and (base or p.Bg) or W
		obj.BackgroundTransparency = th.Transparency
		grad.Color = gradSeq(base or p.Bg, th)
		grad.Rotation = th.GradAngle
		grad.Enabled = not perf
		c.CornerRadius = UDim.new(0, r or th.Corner)
		stroke.Color = perf and p.Rim2 or W
		stroke.Transparency = 1 - th.Stroke / 100
		stroke.Thickness = th.StrokeW
		sg.Color = borderSeq(th, p)
		sg.Enabled = not perf
		if not tracked then
			tracked = true
			window.FlowList[sg] = true
		end
		if th.FlowMode == "关闭" or perf then
			sg.Rotation = 45
		end
	end
	return style, stroke
end

local function extraOf(module)
	if not module.ExtraText then
		return nil
	end
	local ok, res = pcall(module.ExtraText)
	if ok and res ~= nil and tostring(res) ~= "" then
		return tostring(res)
	end
	return nil
end

local function sanitize(name)
	name = tostring(name or ""):gsub("[^%w_%-\128-\255]", "")
	if name == "" then
		name = "default"
	end
	return name
end

local Maid = {}
Maid.__index = Maid

function Maid.new()
	return setmetatable({Items = {}}, Maid)
end

function Maid:Give(item)
	table.insert(self.Items, item)
	return item
end

function Maid:Clean()
	for _, item in self.Items do
		local kind = typeof(item)
		if kind == "RBXScriptConnection" then
			pcall(function()
				item:Disconnect()
			end)
		elseif kind == "Instance" then
			pcall(function()
				item:Destroy()
			end)
		elseif kind == "function" then
			pcall(item)
		end
	end
	table.clear(self.Items)
end

local function Builder(window, parent, P, options, onAdd, onChange)
	local b = {}
	local order = 0

	local function nextOrder()
		order += 1
		return order
	end

	local function added()
		if onAdd then
			onAdd()
		end
	end

	local function changed()
		if onChange then
			onChange()
		end
	end

	local function text(parentObj, content, size, position, textSize)
		return create("TextLabel", {
			Size = size,
			Position = position,
			BackgroundTransparency = 1,
			Text = content,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextSize = textSize or 15,
			Font = T.Font
		}, parentObj)
	end

	local function bindSlide(hit, track, onValue, onActive)
		hit.InputBegan:Connect(function(input)
			if not isPointer(input) then return end
			local guard = window.Theme.SliderGuard and input.UserInputType == Enum.UserInputType.Touch
			local startPos = input.Position
			local active = not guard
			local dead = false
			local function update(pos)
				onValue(math.clamp((pos.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1))
			end
			local function engage(pos)
				active = true
				window:Lock(true)
				if onActive then
					onActive(true)
				end
				update(pos)
			end
			if active then
				engage(input.Position)
			end
			window:StartDrag(input, function(cur)
				if dead then return end
				if not active then
					local dx = math.abs(cur.Position.X - startPos.X)
					local dy = math.abs(cur.Position.Y - startPos.Y)
					if dx >= 6 and dx > dy * 1.2 then
						engage(cur.Position)
					elseif dy >= 8 then
						dead = true
					end
					return
				end
				update(cur.Position)
			end, function()
				if active then
					window:Lock(false)
					if onActive then
						onActive(false)
					end
				end
			end)
		end)
	end

	function b:CreateDivider(content)
		local item = create("Frame", {
			Size = UDim2.new(1, 0, 0, 26),
			BackgroundTransparency = 1,
			LayoutOrder = nextOrder()
		}, parent)
		local label = text(item, content, UDim2.new(1, -28, 1, 0), UDim2.fromOffset(14, 2), 13)
		label.Font = T.FontBold
		local line = create("Frame", {
			Size = UDim2.new(1, -28, 0, 1),
			Position = UDim2.new(0, 14, 1, -3),
			BorderSizePixel = 0
		}, item)
		local lineGrad = create("UIGradient", {}, line)
		window:Reg(function()
			local p = P()
			local style = window.Theme.DividerStyle
			label.TextColor3 = p.Sub
			line.BackgroundColor3 = p.Off
			line.Visible = style ~= "隐藏"
			lineGrad.Transparency = style == "渐隐" and dividerFade or (style == "点线" and dividerDots or dividerNone)
		end)
	end

	function b:CreateToggle(s)
		added()
		local api = {Type = "Toggle", Value = false}
		local item = create("TextButton", {
			Size = UDim2.new(1, 0, 0, 32),
			BackgroundTransparency = 1,
			Text = "",
			AutoButtonColor = false,
			LayoutOrder = nextOrder()
		}, parent)
		local label = text(item, s.Name, UDim2.new(1, -64, 1, 0), UDim2.fromOffset(14, 0))
		local track = create("Frame", {
			Size = UDim2.fromOffset(34, 18),
			Position = UDim2.new(1, -48, 0.5, -9),
			BorderSizePixel = 0
		}, item)
		local trackCorner = corner(track, 9)
		local knob = create("Frame", {
			Size = UDim2.fromOffset(14, 14),
			Position = UDim2.fromOffset(2, 2),
			BorderSizePixel = 0
		}, track)
		local knobCorner = corner(knob, 7)
		local cb = s.Callback or s.Function or function() end
		local function refresh(animate)
			local p = P()
			local on = api.Value
			local square = window.Theme.ToggleStyle == "方块"
			trackCorner.CornerRadius = UDim.new(0, square and 3 or 9)
			knobCorner.CornerRadius = UDim.new(0, square and 2 or 7)
			label.TextColor3 = p.Text
			apply(track, {BackgroundColor3 = on and p.Fill or p.Off}, animate)
			apply(knob, {
				BackgroundColor3 = on and p.OnText or p.Text,
				Size = UDim2.fromOffset(14, 14),
				Position = UDim2.fromOffset(on and 18 or 2, 2)
			}, animate)
		end
		window:Reg(refresh)
		function api:SetValue(state, silent)
			self.Value = state
			refresh(true)
			if not silent then
				window:Safe(s.Name, cb, state)
				changed()
			end
		end
		function api:Save()
			return self.Value
		end
		function api:Load(v)
			if type(v) == "boolean" and v ~= self.Value then
				self:SetValue(v)
			end
		end
		item.ClipsDescendants = true
		item.InputBegan:Connect(function(input)
			if isPointer(input) then
				window:Ripple(item, input)
				local th = window.Theme
				if th.Micro and th.Animations and not th.Perf then
					tween(knob, {Size = UDim2.fromOffset(20, 14), Position = UDim2.fromOffset(api.Value and 12 or 2, 2)}, 0.1)
				end
			end
		end)
		item.InputEnded:Connect(function(input)
			if isPointer(input) then
				refresh(true)
			end
		end)
		item.MouseButton1Click:Connect(function()
			api:SetValue(not api.Value)
		end)
		if s.Default then
			api:SetValue(true, true)
		end
		if options then
			options[s.Name] = api
		end
		return api
	end

	function b:CreateSlider(s)
		added()
		local min, max = s.Min or 0, s.Max or 100
		local step = s.Step or 1
		local api = {Type = "Slider", Value = s.Default or min}
		local item = create("Frame", {
			Size = UDim2.new(1, 0, 0, 46),
			BackgroundTransparency = 1,

local function luminance(c)
	return 0.299 * c.R + 0.587 * c.G + 0.114 * c.B
end

local function linearChannel(x)
	if x <= 0.03928 then
		return x / 12.92
	end
	return ((x + 0.055) / 1.055) ^ 2.4
end

local function relLum(c)
	return 0.2126 * linearChannel(c.R) + 0.7152 * linearChannel(c.G) + 0.0722 * linearChannel(c.B)
end

local function contrastRatio(a, b)
	local l1, l2 = relLum(a), relLum(b)
	if l1 < l2 then
		l1, l2 = l2, l1
	end
	return (l1 + 0.05) / (l2 + 0.05)
end

local function ensureContrast(fg, bg, minRatio)
	if contrastRatio(fg, bg) >= minRatio then
		return fg
	end
	local target = relLum(bg) > 0.4 and B or W
	for i = 1, 10 do
		local c = fg:Lerp(target, i / 10)
		if contrastRatio(c, bg) >= minRatio then
			return c
		end
	end
	return target

local pinyinMap

local function pinyinOf(str)
	if not pinyinMap then
		pinyinMap = {}
		local i = 0
		for _, cp in utf8.codes(PinyinChars) do
			i += 1
			pinyinMap[utf8.char(cp)] = string.sub(PinyinInitials, i, i)
		end
	end
	local ok, result = pcall(function()
		local out = {}
		for _, cp in utf8.codes(str) do
			local ch = utf8.char(cp)
			out[#out + 1] = pinyinMap[ch] or string.lower(ch)
		end
		return table.concat(out)
	end)
	return ok and result or string.lower(str)
end

function Library:BuildPals()
	local th = self.Theme
	local tint = Tints[th.Tint]
	if th.Tint == "自定义" then
		local ok, c = pcall(Color3.fromHex, th.AccentHex)
		tint = ok and c or nil
	end
	local mix = math.clamp(th.TintMix / 100, 0, 1)
	local function ov(key)
		local h = th[key]
		if type(h) == "string" and h ~= "" then
			local ok, c = pcall(Color3.fromHex, h)
			if ok then
				return c
			end
		end
	end
	local function tinted(base)
		local p = {}
		for k, v in base do
			p[k] = v
		end
		if tint then
			p.RowOn = p.RowOn:Lerp(tint, mix)
			p.Fill = p.Fill:Lerp(tint, mix)
			p.Rim1 = p.Rim1:Lerp(tint, math.clamp(mix * 1.6, 0, 1))
			if luminance(p.RowOn) > 0.55 then
				p.OnText = Color3.fromRGB(10, 10, 12)
			else
				p.OnText = Color3.fromRGB(244, 244, 248)
			end
		end
		local o = ov("ColBg")
		if o then p.Bg = o end
		o = ov("ColHead")
		if o then p.Head = o end
		o = ov("ColRow")
		if o then p.Row = o end
		o = ov("ColText")
		if o then p.Text = o end
		o = ov("ColSub")
		if o then p.Sub = o end
		o = ov("ColRowOn")
		if o then
			p.RowOn = o
			p.OnText = luminance(o) > 0.55 and Color3.fromRGB(10, 10, 12) or Color3.fromRGB(244, 244, 248)
		end
		o = ov("ColFill")
		if o then p.Fill = o end
		o = ov("ColRim")
		if o then
			p.Rim1 = o
			p.Rim2 = p.Rim2:Lerp(o, 0.5)
		end
		if th.AutoContrast then
			for _, bg in {p.Bg, p.Row} do
				p.Text = ensureContrast(p.Text, bg, 4.5)
			end
			for _, bg in {p.Bg, p.Row, p.Set} do
				p.Sub = ensureContrast(p.Sub, bg, 3)
			end
			p.OnText = ensureContrast(p.OnText, p.RowOn, 4.5)
			p.OnText = ensureContrast(p.OnText, p.Fill, 3)
		end
		p.HeadText = p.Text
		p.HeadSub = p.Sub
		if th.AutoContrast then
			p.HeadText = ensureContrast(p.Text, p.Head, 4.5)
			p.HeadSub = ensureContrast(p.Sub, p.Head, 3)
		end
		p.Accent = p.Fill
		p.Tinted = tint ~= nil
		p.TintColor = tint
		return p
	end
	self.Pals = {Dark = tinted(DARK), Light = tinted(LIGHT)}
end

function Library:ContrastReport()
	local mode = self.Theme.Mode
	local list = {}
	if mode ~= "亮白" then
		table.insert(list, self.Pals.Dark)
	end
	if mode ~= "深黑" then
		table.insert(list, self.Pals.Light)
	end
	local worst = 99
	for _, p in list do
		worst = math.min(worst, contrastRatio(p.Text, p.Bg), contrastRatio(p.HeadText, p.Head), contrastRatio(p.Text, p.Row), contrastRatio(p.Sub, p.Bg), contrastRatio(p.OnText, p.RowOn))
	end
	return worst
end

function Library:Palette(index)
	local mode = self.Theme.Mode
	local dark
	if mode == "深黑" then
		dark = true
	elseif mode == "亮白" then
		dark = false
	else
		dark = index % 2 == 1
	end
	return dark and self.Pals.Dark or self.Pals.Light
end

function Library:Reg(fn)
	table.insert(self.Regs, fn)
	fn(false)
	return fn
end

function Library:Unreg(fn)
	local i = table.find(self.Regs, fn)
	if i then
		table.remove(self.Regs, i)
	end
end

function Library:Safe(name, fn, ...)
	local args = table.pack(...)
	task.spawn(function()
		local ok, err = pcall(fn, table.unpack(args, 1, args.n))
		if not ok then
			warn("[ClickGui] " .. tostring(name) .. ": " .. tostring(err))
			self:Notify("错误", tostring(name) .. "：" .. tostring(err):sub(1, 50), 3, "alert")
		end
	end)
end

function Library:StartDrag(input, onMove, onEnd)
	local drag = {Input = input, OnMove = onMove, OnEnd = onEnd}
	self.Drags[drag] = true
	return drag
end

function Library:EndDrag(drag)
	self.Drags[drag] = nil
end

function Library:MakeDraggable(handle, target, opts)
	opts = opts or {}
	handle.InputBegan:Connect(function(input)
		if not isPointer(input) then return end
		if opts.OnDown then
			opts.OnDown(input)
		end
		local startPos = input.Position
		local origin = target.Position
		local moved = false
		local finished = false
		local holdFired = false
		local drag
		drag = self:StartDrag(input, function(cur)
			local d = cur.Position - startPos
			if d.Magnitude > 8 then
				moved = true
			end
			if moved and not opts.NoMove and (not opts.CanMove or opts.CanMove()) then
				local s = opts.GetScale and opts.GetScale() or 1
				target.Position = UDim2.new(origin.X.Scale, origin.X.Offset + d.X / s, origin.Y.Scale, origin.Y.Offset + d.Y / s)
			end
		end, function()
			finished = true
			if opts.OnUp then
				opts.OnUp()
			end
			if moved then
				if opts.OnMoved then
					opts.OnMoved()
				end
			elseif not holdFired and opts.OnTap then
				opts.OnTap()
			end
		end)
		if opts.OnHold then
			task.delay(opts.HoldTime or 0.7, function()
				if not finished and not moved then
					finished = true
					holdFired = true
					self:EndDrag(drag)
					if opts.OnUp then
						opts.OnUp()
					end
					opts.OnHold()
				end
			end)
		end
	end)
end

function Library:SetUIScale(value)
	value = math.clamp(math.floor(value + 0.5), 40, 200)
	if value == self.Theme.UIScale then return end
	self.Theme.UIScale = value
	self:UpdateScale()
	local c = self.Controls.UIScale
	if c then
		c:SetValue(value, true)
	end
end

function Library:UpdateGrip()
	self.Grip.Visible = self.Visible and self.Theme.ShowGrip
end

function Library:Ripple(host, input, color)
	local th = self.Theme
	if not th.Ripple or th.Perf then return end
	local s = self.Scale.Scale
	local inset = GuiService:GetGuiInset().Y
	local x = (input.Position.X - host.AbsolutePosition.X) / s
	local y = (input.Position.Y + inset - host.AbsolutePosition.Y) / s
	local p = self:Palette(1)
	local dot = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(x, y),
		Size = UDim2.fromOffset(0, 0),
		BackgroundColor3 = color or p.Text,
		BackgroundTransparency = 0.82,
		BorderSizePixel = 0,
		ZIndex = 0
	}, host)
	corner(dot, 999)
	local size = math.max(host.AbsoluteSize.X, host.AbsoluteSize.Y) / s * 2.2
	tween(dot, {Size = UDim2.fromOffset(size, size), BackgroundTransparency = 1}, 0.5)
	task.delay(0.55, function()
		dot:Destroy()
	end)
end

function Library:Wave(host, x, color)
	local th = self.Theme
	if not th.Wave or th.Perf or not th.Animations or not self.IsReady or not self.Visible then return end
	local s = self.Scale.Scale
	local w = host.AbsoluteSize.X / s
	local h = host.AbsoluteSize.Y / s
	x = x or w / 2
	local far = math.max(x, w - x)
	local d = math.sqrt(far * far + (h / 2) * (h / 2)) * 2.1
	local dot = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromOffset(x, h / 2),
		Size = UDim2.fromOffset(0, 0),
		BackgroundColor3 = color,
		BackgroundTransparency = 0.45,
		BorderSizePixel = 0,
		ZIndex = 0
	}, host)
	corner(dot, 999)
	tween(dot, {Size = UDim2.fromOffset(d, d)}, 0.38)
	task.delay(0.3, function()
		tween(dot, {BackgroundTransparency = 1}, 0.2)
	end)
	task.delay(0.56, function()
		dot:Destroy()
	end)
end

function Library:SetGlow(cat, on)
	if cat.GlowOn == on then return end
	cat.GlowOn = on
	local th = self.Theme
	local target = on and (1 - (th.GlowStrength / 100) * 0.07) or 1
	if on then
		cat.Glow.Visible = true
	end
	for _, r in cat.GlowRings do
		apply(r, {BackgroundTransparency = target}, true)
	end
	if not on then
		task.delay(0.28, function()
			if not cat.GlowOn then
				cat.Glow.Visible = false
			end
		end)
	end
end

function Library:UpdateGlow(pos, down)
	local th = self.Theme
	local show = down and th.TouchGlow and not th.Perf and self.Visible
	local inset = GuiService:GetGuiInset().Y
	local s = self.Scale.Scale
	for _, cat in self.Categories do
		if cat.Glow then
			local on = false
			if show then
				local pp, ps = cat.Panel.AbsolutePosition, cat.Panel.AbsoluteSize
				local x, y = pos.X - pp.X, pos.Y + inset - pp.Y
				if x >= 0 and y >= 0 and x <= ps.X and y <= ps.Y then
					cat.Glow.Position = UDim2.fromOffset(x / s, y / s)
					on = true
				end
			end
			self:SetGlow(cat, on)
		end
	end
end

function Library:ShimmerStep(clock)
	local sps = 0.15 + (self.Theme.ShimmerSpeed / 100) * 0.55
	for _, m in self.Modules do
		if m.Enabled then
			local x = -0.4 + ((clock * sps + m.ShineOffset) % 2.2)
			m.Shine.Position = UDim2.new(x, 0, 0.5, 0)
		end
	end
end

function Library:UpdateTicker()
	local th = self.Theme
	local need = next(self.Rainbows) ~= nil or th.HudColor == "彩虹" or ((th.Shimmer or th.AccentRainbow or th.FlowMode ~= "关闭") and not th.Perf)
	if need and not self.Ticker then
		local acc = 0
		self.Ticker = RunService.Heartbeat:Connect(function(dt)
			acc += dt
			if acc < 1 / 30 then return end
			acc = 0
			local clock = os.clock()
			self.Hue = (clock * 0.2) % 1
			for api in self.Rainbows do
				api:SetHSV(self.Hue, nil, nil, false, true)
			end
			if th.HudColor == "彩虹" then
				self:HudColorize()
			end
			if not th.Perf then
				if th.FlowMode ~= "关闭" then
					local deg = (clock * th.FlowSpeed) % 360
					for g in self.FlowList do
						if g.Parent == nil then
							self.FlowList[g] = nil
						else
							g.Rotation = deg
						end
					end
				end
				if th.AccentRainbow then
					for _, m in self.Modules do
						if m.Enabled then
							m.Refresh(false)
						end
					end
				end
				if th.Shimmer then
					self:ShimmerStep(clock)
				end
			end
		end)
		self.Maid:Give(self.Ticker)
	elseif not need and self.Ticker then
		self.Ticker:Disconnect()
		self.Ticker = nil
	end
end

end

local FontSets = {
	["默认"] = {Enum.Font.SourceSans, Enum.Font.SourceSansBold},
	["Gotham"] = {Enum.Font.Gotham, Enum.Font.GothamBold},
	["Arial"] = {Enum.Font.Arial, Enum.Font.ArialBold}
}

function Library:ApplyFont()
	local name = self.Theme.FontName
	if self.FontApplied == name then return end
	local set = FontSets[name] or FontSets["默认"]
	local oldRegular, oldBold = T.Font, T.FontBold
	T.Font, T.FontBold = set[1], set[2]
	self.FontApplied = name
	if oldRegular == T.Font then return end
	for _, d in self.Gui:GetDescendants() do
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			local ok, f = pcall(function()
				return d.Font
			end)
			if ok then
				if f == oldBold then
					d.Font = T.FontBold
				elseif f == oldRegular then
					d.Font = T.Font
				end
			end
		end
	end
end

function Library:TrackHistory()
	if self.Restoring or self.Loading or not self.HistLast then
		self.HistLast = table.clone(self.Theme)
		return
	end
	if self.Previewing then return end
	local diff = false
	for k, v in self.Theme do
		if self.HistLast[k] ~= v then
			diff = true
			break
		end
	end
	if not diff then return end
	local now = os.clock()
	if now - (self.HistTime or -10) > 0.8 then
		table.insert(self.Undo, self.HistLast)
		if #self.Undo > 30 then
			table.remove(self.Undo, 1)
		end
	end
	self.HistTime = now
	self.Redo = {}
	self.HistLast = table.clone(self.Theme)
end

function Library:RestoreTheme(snap)
	self.Restoring = true
	for k, v in snap do
		self.Theme[k] = v
	end
	self:SyncControls()
	self:ApplyTheme()
	self.Restoring = false
	self.HistLast = table.clone(self.Theme)
	self.HistTime = -10
	self:QueueSave()
end

function Library:UndoTheme()
	self:TrackHistory()

		self:Notify("历史", "没有可撤销的外观改动", 1.4, "warning")
		return false
	end
	table.insert(self.Redo, table.clone(self.Theme))
	self:RestoreTheme(snap)
	return true
end

function Library:SpectrumColor(obj, module)
	local th = self.Theme
	local mode = th.EnabledColor
	local root = self.Root
	local t = 0
	if root and root.AbsoluteSize.Y > 0 then
		t = math.clamp((obj.AbsolutePosition.Y - root.AbsolutePosition.Y) / root.AbsoluteSize.Y, 0, 1)
	end
	local function hex(h, default)
		local ok, c = pcall(Color3.fromHex, h)
		return ok and c or default
	end
	if mode == "日落" then
		return Color3.fromRGB(255, 96, 96):Lerp(Color3.fromRGB(255, 196, 70), t)
	elseif mode == "双色渐变" then
		return hex(th.EnabledHexA, W):Lerp(hex(th.EnabledHexB, W), t)
	elseif mode == "顺序渐变" and module then
		local cat = self:FindCategory(module.Host and module.Host.Name or module.Category)
		local count = cat and #cat.Modules or 1
		local idx = cat and table.find(cat.Modules, module) or 1
		return hex(th.EnabledHexA, W):Lerp(hex(th.EnabledHexB, W), (idx - 1) / math.max(count - 1, 1))
	elseif mode == "分类取色" and module then
		local cat = self:FindCategory(module.Host and module.Host.Name or module.Category)
		if cat and cat.ColorHex then
			local ok, c = pcall(Color3.fromHex, cat.ColorHex)
			if ok then
				return c
			end
		end
		local n = math.max(#self.Categories, 1)
		return Color3.fromHSV(((cat and cat.Index or 1) - 1) / n % 1, 0.62, 0.92)
	end
	return Color3.fromHSV(0.3 + t * 0.48, 0.72, 0.88)
end

function Library:ExportTheme()
	local data = {Preset = self.Theme.Preset}
	for _, k in VisualKeys do
		data[k] = self.Theme[k]
	end
	return HttpService:JSONEncode(data)
end

function Library:ImportTheme(code)
	local ok, data = pcall(function()
		return HttpService:JSONDecode(code)
	end)
	if not ok or type(data) ~= "table" then
		return false
	end
	local count = 0
	for _, k in VisualKeys do
		local v = data[k]
		if v ~= nil and type(v) == type(DefaultTheme[k]) then
			self.Theme[k] = v
			count += 1
		end
	end
	if count == 0 then
		return false
	end
	if type(data.Preset) == "string" then
		self.Theme.Preset = data.Preset
	end
	self:SyncControls()
	self:ApplyTheme()
	self:QueueSave()
	return true
end

function Library:ThemePath(name)
	return self.Folder .. "/theme_" .. name .. ".json"
end

function Library:CleanName(name)
	name = tostring(name or ""):gsub('[/\\:*?"<>|]', ""):gsub("^%s+", ""):gsub("%s+$", "")
	return name
end

function Library:ListThemes()
	local list = {}
	if self.CanFile and listfiles then
		local ok, files = pcall(listfiles, self.Folder)
		if ok and type(files) == "table" then
			for _, path in files do
				local base = tostring(path):match("([^/\\]+)%.json$")
				if base and base:sub(1, 6) == "theme_" and #base > 6 then
					local name = base:sub(7)
					if not table.find(list, name) then
						table.insert(list, name)
					end
				end
			end
		end
	end
	table.sort(list)
	return list
end

function Library:SaveTheme(name)
	name = self:CleanName(name)
	if name == "" then
		self:Notify("我的主题", "请先输入主题名称", 2, "warning")
		return false
	end
	if not self.CanFile then
		self:Notify("我的主题", "当前环境不支持文件操作", 2.5, "warning")
		return false
	end
	local ok = pcall(function()
		if not isfolder(self.Folder) then
			makefolder(self.Folder)
		end
		writefile(self:ThemePath(name), self:ExportTheme())
	end)
	if ok then
		self:Notify("我的主题", "已保存：" .. name, 1.8, "info")
	else
		self:Notify("我的主题", "保存失败", 2.5, "alert")
	end
	return ok, name
end

function Library:LoadTheme(name)
	if not self.CanFile or not name then return false end
	local ok, content = pcall(function()
		local path = self:ThemePath(name)
		if isfile(path) then
			return readfile(path)
		end
	end)
	if ok and content and self:ImportTheme(content) then
		self:Notify("我的主题", "已应用：" .. name, 1.8, "info")
		return true
	end
	self:Notify("我的主题", "找不到或无法读取：" .. tostring(name), 2.2, "warning")
	return false
end

function Library:DeleteTheme(name)
	if not self.CanFile or not delfile or not name then return false end
	local ok = pcall(function()
		local path = self:ThemePath(name)
		if isfile(path) then
			delfile(path)
		end
	end)
	if ok then
		self:Notify("我的主题", "已删除：" .. name, 1.6, "info")
	end
	return ok
end

function Library:BuildCape()
	self:DestroyCape()
	local th = self.Theme
	if not th.Cape then return end
	local lp = Players.LocalPlayer
	local char = lp and lp.Character
	if not char then return end
	local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
	local root = char:FindFirstChild("HumanoidRootPart")
	if not torso or not root then return end
	local len = th.CapeLen
	local id = tostring(th.CapeId or "")
	local part = create("Part", {
		Name = randomName(),
		Size = Vector3.new(th.CapeW, len, 0.12),
		Color = hexColor(th.CapeHex, Color3.fromRGB(20, 20, 26)),
		Material = Enum.Material.Fabric,
		CanCollide = false,
		CanTouch = false,
		CanQuery = false,
		Massless = true
	})
	create("Decal", {
		Face = Enum.NormalId.Back,
		Texture = id:match("^%d+$") and ("rbxassetid://" .. id) or Assets.Vape
	}, part)
	local base = CFrame.new(0, torso.Size.Y / 2 - 0.1, torso.Size.Z / 2 + 0.06)
	local motor = create("Motor6D", {
		Name = "CapeMotor",
		Part0 = torso,
		Part1 = part,
		C0 = base,
		C1 = CFrame.new(0, len / 2, 0)
	}, part)
	part.Parent = char
	local data = {Part = part, Sig = self:CapeSig(), Smooth = 6, Phase = 0}
	data.Conn = RunService.Heartbeat:Connect(function(dt)
		if not part.Parent or not root.Parent then return end
		local v = root.AssemblyLinearVelocity
		local look = root.CFrame.LookVector
		local right = root.CFrame.RightVector
		local forward = v.X * look.X + v.Z * look.Z
		local side = v.X * right.X + v.Z * right.Z
		local k = self.Theme.CapeSwing / 100
		data.Phase += dt
		local target = 6 + math.clamp(forward, 0, 40) * 1.6 * k + math.clamp(-v.Y, 0, 60) * 0.25 * k
		data.Smooth += (target - data.Smooth) * math.min(dt * 8, 1)
		local sway = math.sin(data.Phase * 2.4) * (2 + math.abs(forward) * 0.15) * k + math.clamp(side, -30, 30) * 0.5 * k
		motor.C0 = base * CFrame.Angles(-math.rad(math.clamp(data.Smooth, 0, 85)), 0, math.rad(sway))
	end)
	self.CapeData = data
end

function Library:UpdateCape()
	local th = self.Theme
	local c = self.CapeData
	if not th.Cape then
		if c then
			self:DestroyCape()
		end
		return
	end
	if not c or c.Sig ~= self:CapeSig() or not c.Part.Parent then
		self:BuildCape()
	end
end

function Library:SetModCustom(key, field, value)
	local m = self.ModulesByKey[key]
	if not m then return false end
	if field == "Name" then
		value = tostring(value or "")
		m.Display = value ~= "" and value or nil
	elseif field == "Color" then
		value = tostring(value or ""):gsub("#", ""):gsub("%s", "")
		if value == "" then
			m.ColorHex = nil
			m.ColorC = nil
		else
			local ok, c = pcall(Color3.fromHex, value)
			if not ok then
				return false
			end
			m.ColorHex = value
			m.ColorC = c
		end
	elseif field == "Icon" then
		local id = tostring(value or ""):gsub("%s", "")
		if id == "" then
			m.IconId = nil
		elseif id:match("^%d+$") then
			m.IconId = id
		else
			return false
		end
	elseif field == "Hidden" then
		m.Hidden = value == true
	elseif field == "Pin" then
		m.Pinned = value == true
	else
		return false
	end
	m:ApplyCustom()
	local quick = self.Quicks[key]
	if quick then
		quick.Refresh(false)
	end
	self:RefreshFilter()
	self:RefreshHud()
	self:QueueSave()
	return true
end

function Library:RefreshModList()
	if not self.ModDrop then return end
	local names = {}
	for _, m in self.Modules do
		table.insert(names, m.Key)
	end
	if #names == 0 then
		names = {"（无）"}
	end
	self.ModDrop:SetList(names)
	if not self.ModSel or not table.find(names, self.ModSel) then
		self.ModSel = names[1]
	end
	self.ModDrop:SetValue(self.ModSel, true)
	self:SyncModControls()
end

function Library:SyncModControls()
	local c = self.ModControls
	local m = self.ModSel and self.ModulesByKey[self.ModSel]
	if not c or not m then return end
	c.Name:SetValue(m.Display or "", true)
	c.Color:SetValue(m.ColorHex or "", true)
	c.Icon:SetValue(m.IconId or "", true)
	c.Pin:SetValue(m.Pinned == true, true)
	c.Hidden:SetValue(m.Hidden == true, true)
end

function Library:RaiseCat(cat)
	self.TopZ = (self.TopZ or 10) + 1
	cat.Holder.ZIndex = self.TopZ

function Library:ClampCat(cat)
	local s = self.Scale.Scale
	local holder = cat.Holder
	local pos = holder.Position
	local maxX = self.Root.AbsoluteSize.X / s - 60
	local maxY = self.Root.AbsoluteSize.Y / s - 44
	local x = math.clamp(pos.X.Offset, 60 - holder.AbsoluteSize.X / s, maxX)
	local y = math.clamp(pos.Y.Offset, 48, math.max(maxY, 48))
	holder.Position = UDim2.fromOffset(x, y)
	cat.FreePos = {X = x, Y = y}
end

function Library:ApplyFreeLayout()

	end
end

function Library:CreateCustomCategory(name, quiet)
	name = tostring(name or "")
	name = name:gsub("^%s+", "")
	name = name:gsub("%s+$", "")
	if name == "" or utf8.len(name) == nil or utf8.len(name) > 10 then
		return false, "名称为空或超过 10 个字"
	end
	for _, c in self.Categories do
		if c.Name == name then
			return false, "已经有同名分类"
		end
	end
	local cat = self:CreateCategory({Name = name})
	cat.Custom = true
	self:ApplyCatLayout()
	self:ApplyFreeLayout()
	if not quiet then
		self:QueueSave()
	end
	return cat
end

function Library:MoveModule(key, targetName, quiet)
	local m = self.ModulesByKey[key]
	if not m then return false end
	local target = (targetName == nil or targetName == "（原位置）") and self:FindCategory(m.Category) or self:FindCategory(targetName)
	if not target then return false end
	local from = m.Host or self:FindCategory(m.Category)
	if from == target then return true end
	if from then
		local i = table.find(from.Modules, m)
		if i then
			table.remove(from.Modules, i)
		end
	end
	table.insert(target.Modules, m)
	m.Container.Parent = target.Body
	m.Host = target.Name ~= m.Category and target or nil
	m.Index = #target.Modules
	m:ApplyCustom()
	if not quiet then
		self:ApplyTheme()
		self:RefreshFilter()
		self:QueueSave()
	end
	return true
end

function Library:DeleteCustomCategory(name, quiet)
	local cat = self:FindCategory(name)
	if not cat or not cat.Custom then return false end
	for _, m in table.clone(cat.Modules) do
		self:MoveModule(m.Key, nil, true)
	end
	local i = table.find(self.Categories, cat)
	if i then
		table.remove(self.Categories, i)
	end
	cat.Holder:Destroy()
	self:ApplyCatLayout()
	self:RefreshCatList()
	if not quiet then
		self:ApplyTheme()
		self:RefreshFilter()
		self:QueueSave()
	end
	return true
end

function Library:RenameCustomCategory(old, new)
	local cat = self:FindCategory(old)
	if not cat or not cat.Custom then return false, "只能重命名自定义分类" end
	new = tostring(new or "")
	new = new:gsub("^%s+", "")
	new = new:gsub("%s+$", "")
	if new == "" or utf8.len(new) == nil or utf8.len(new) > 10 then
		return false, "名称为空或超过 10 个字"
	end
	if new ~= old and self:FindCategory(new) then
		return false, "已经有同名分类"
	end
	cat.Name = new
	cat.Holder.Name = new
	cat.TitleLabel.Text = new
	if self.CatSel == old then
		self.CatSel = new
	end
	self:RefreshCatList()
	self:QueueSave()
	return true
end

function Library:ApplyCustomCats(info)
	local wanted = {}
	if type(info.Names) == "table" then
		for _, n in info.Names do
			if type(n) == "string" then
				wanted[n] = true
			end
		end
	end
	for _, cat in table.clone(self.Categories) do
		if cat.Custom and not wanted[cat.Name] then
			self:DeleteCustomCategory(cat.Name, true)
		end
	end
	if type(info.Names) == "table" then
		for _, n in info.Names do
			if type(n) == "string" and not self:FindCategory(n) then
				self:CreateCustomCategory(n, true)
			end
		end
	end
	local assigned = {}
	if type(info.Assign) == "table" then
		for _, a in info.Assign do
			if type(a) == "table" and type(a.Key) == "string" and type(a.Cat) == "string" then
				assigned[a.Key] = true
			end
		end
	end
	for _, m in table.clone(self.Modules) do
		if m.Host and not assigned[m.Key] then
			self:MoveModule(m.Key, nil, true)
		end
	end
	if type(info.Assign) == "table" then
		for _, a in info.Assign do
			if type(a) == "table" and type(a.Key) == "string" and type(a.Cat) == "string" then
				self:MoveModule(a.Key, a.Cat, true)
			end
		end
	end
end

function Library:RefreshMoveList()
	if not self.MoveDrop then return end
	local list = {"（原位置）"}
	for _, cat in self.Categories do
		if not cat.Floating then
			table.insert(list, cat.Name)
		end
	end
	self.MoveDrop:SetList(list)
	if not self.MoveSel or not table.find(list, self.MoveSel) then
		self.MoveSel = list[1]
	end
	self.MoveDrop:SetValue(self.MoveSel, true)
end

function Library:FindCategory(name)
	for _, cat in self.Categories do
		if cat.Name == name and not cat.Floating then
			return cat
		end
	end
end

function Library:ApplyCatLayout()
	local list = {}
	for _, cat in self.Categories do
		if not cat.Floating then
			table.insert(list, cat)
		end
	end
	table.sort(list, function(a, b)
		if a.Order == b.Order then
			return a.Index < b.Index
		end
		return a.Order < b.Order
	end)
	for i, cat in list do
		cat.Order = i
		cat:ApplyStyle()
	end
	return list
end

function Library:SetCatStyle(name, key, value)
	local cat = self:FindCategory(name)
	if not cat then return false end
	if key == "Color" then
		value = tostring(value or ""):gsub("#", ""):gsub("%s", "")
		if value == "" then
			cat.ColorHex = nil
		else
			local ok = pcall(Color3.fromHex, value)
			if not ok then
				return false
			end
			cat.ColorHex = value
		end
		self:ApplyTheme()
	elseif key == "Icon" then
		value = tostring(value or ""):gsub("%s", "")
		if value == "" then
			cat.IconId = nil
		elseif value:match("^%d+$") then
			cat.IconId = value
		else
			return false
		end
		cat:ApplyStyle()
	elseif key == "Width" then
		cat.WidthOverride = type(value) == "number" and math.clamp(value, 120, 320) or nil
		cat:ApplyStyle()
	elseif key == "Hidden" then
		cat.Hidden = value == true
		cat:ApplyStyle()
	end
	self:QueueSave()
	return true
end

function Library:MoveCat(name, dir)
	local list = self:ApplyCatLayout()
	for i, cat in list do
		if cat.Name == name then
			local other = list[i + dir]
			if other then
				cat.Order, other.Order = other.Order, cat.Order
				self:ApplyCatLayout()
				self:QueueSave()
				return true
			end
			return false
		end
	end
	return false
end

function Library:QueueTheme()
	if self.ThemePending then return end
	self.ThemePending = true
	task.delay(0.06, function()
		self.ThemePending = false
		if not self.Destroyed then
			self:ApplyTheme()
		end
	end)
end

function Library:QueueSave()
	if not self.IsReady or self.Loading or self.Previewing or not self.Meta.AutoSave then return end
	self.SaveToken += 1
	local token = self.SaveToken
	task.delay(1.2, function()
		if token == self.SaveToken and not self.Destroyed then
			self:SaveConfig(self.Meta.Config, true)
		end
	end)
end

function Library:Serialize()
	local data = {Version = 5, Theme = {}, Modules = {}, Categories = {}, Quick = {}}
	for k, v in self.Theme do
		data.Theme[k] = v
	end
	for key, module in self.ModulesByKey do
		local entry = {Enabled = module.Enabled, Options = {}, Custom = {Name = module.Display or "", Icon = module.IconId or "", Color = module.ColorHex or "", Hidden = module.Hidden == true, Pin = module.Pinned == true}}
		for name, opt in module.Options do
			if opt.Save then
				entry.Options[name] = opt:Save()
			end
		end
		data.Modules[key] = entry
	end
	for _, cat in self.Categories do
		if not cat.Floating then
			data.Categories[cat.Name] = {Collapsed = cat.Collapsed, Color = cat.ColorHex or "", Width = cat.WidthOverride or 0, Hidden = cat.Hidden == true, Order = cat.Order, Icon = cat.IconId or "", X = cat.FreePos and cat.FreePos.X, Y = cat.FreePos and cat.FreePos.Y}
		end
	end
	for key, q in self.Quicks do
		data.Quick[key] = {X = q.X, Y = q.Y}
	end
	local names, assigns = {}, {}
	for _, cat in self.Categories do
		if not cat.Floating then
			if cat.Custom then
				table.insert(names, cat.Name)
			end
			for _, m in cat.Modules do
				if m.Host then
					table.insert(assigns, {Key = m.Key, Cat = cat.Name})
				end
			end
		end
	end
	data.CustomCats = {Names = names, Assign = assigns}
	return data
end

function Library:SyncControls()
	local th = self.Theme
	for key, api in self.Controls do
		if api.Type == "Color" then
			local ok, c = pcall(Color3.fromHex, th[api.ThemeKey])
			if ok then
				local h, sat, val = c:ToHSV()
				api:SetHSV(h, sat, val, true, true)
			end
		elseif key == "Transparency" then
			api:SetValue(math.floor(th.Transparency * 100 + 0.5), true)
		else
			api:SetValue(th[key], true)
		end
	end
end

function Library:Deserialize(data)
	self.Loading = true
	if type(data.CustomCats) == "table" then
		self:ApplyCustomCats(data.CustomCats)
	end
	if type(data.Theme) == "table" then
		for k, v in data.Theme do
			if DefaultTheme[k] ~= nil and type(v) == type(DefaultTheme[k]) then
				self.Theme[k] = v
			end
		end
	end
	self:SyncControls()
	self:ApplyTheme()
	if type(data.Modules) == "table" then
		for key, entry in data.Modules do
			local module = self.ModulesByKey[key]
			if module and type(entry) == "table" then
				if type(entry.Custom) == "table" then
					local cu = entry.Custom
					module.Display = type(cu.Name) == "string" and cu.Name ~= "" and cu.Name or nil
					module.ColorHex = nil
					module.ColorC = nil
					if type(cu.Color) == "string" and cu.Color ~= "" then
						local okc, cc = pcall(Color3.fromHex, cu.Color)
						if okc then
							module.ColorHex = cu.Color
							module.ColorC = cc
						end
					end
					module.IconId = type(cu.Icon) == "string" and cu.Icon:match("^%d+$") or nil
					module.Hidden = cu.Hidden == true
					module.Pinned = cu.Pin == true
					module:ApplyCustom()
				end
				if type(entry.Options) == "table" then
					for name, value in entry.Options do
						local opt = module.Options[name]
						if opt and opt.Load then
							pcall(opt.Load, opt, value)
						end
					end
				end
				local state = entry.Enabled == true
				if module.Enabled ~= state then
					module:Toggle(state, true)
				end
			end
		end
	end
	if type(data.Categories) == "table" then
		for _, cat in self.Categories do
			local saved = data.Categories[cat.Name]
			if saved and type(saved.Collapsed) == "boolean" then
				cat:SetCollapsed(saved.Collapsed)
			end
			if saved and not cat.Floating then
				local hex = type(saved.Color) == "string" and saved.Color or ""
				local okc = hex ~= "" and pcall(Color3.fromHex, hex)
				cat.ColorHex = okc and hex or nil
				cat.WidthOverride = type(saved.Width) == "number" and saved.Width >= 120 and saved.Width or nil
				cat.Hidden = saved.Hidden == true
				cat.FreePos = (type(saved.X) == "number" and type(saved.Y) == "number") and {X = saved.X, Y = saved.Y} or nil
				cat.IconId = type(saved.Icon) == "string" and saved.Icon:match("^%d+$") or nil
				if type(saved.Order) == "number" then
					cat.Order = saved.Order
				end
			end
		end
		self:ApplyCatLayout()
		self:ApplyFreeLayout()
		self:ApplyTheme()
		self:SyncCatControls()
	end
	if type(data.Quick) == "table" then
		for key in table.clone(self.Quicks) do
			local module = self.ModulesByKey[key]
			if module then
				self:RemoveQuick(module, true)
			end
		end
		for key, q in data.Quick do
			local module = self.ModulesByKey[key]
			if module and type(q) == "table" and type(q.X) == "number" and type(q.Y) == "number" then
				self:CreateQuick(module, q.X, q.Y)
			end
		end
	end
	self:RefreshFilter()
	self:RefreshHud()
	self.Loading = false
end

function Library:ListConfigs()
	local list = {}
	if self.CanFile and listfiles then
		local ok, files = pcall(listfiles, self.Folder)
		if ok and type(files) == "table" then
			for _, path in files do
				local name = tostring(path):match("([^/\\]+)%.json$")
				if name and name ~= "_meta" and name:sub(1, 6) ~= "theme_" and name:sub(1, 4) ~= "bak_" and not table.find(list, name) then
					table.insert(list, name)
				end
			end
		end
	end
	if not table.find(list, "default") then
		table.insert(list, "default")
	end
	if self.Meta.Config and not table.find(list, self.Meta.Config) then
		table.insert(list, self.Meta.Config)
	end
	table.sort(list)
	return list
end

function Library:LoadMeta()
	if not self.CanFile then return end
	pcall(function()
		local path = self.Folder .. "/_meta.json"
		if isfile(path) then
			local data = HttpService:JSONDecode(readfile(path))
			if type(data) == "table" then
				if type(data.Config) == "string" then
					self.Meta.Config = sanitize(data.Config)
				end
				if type(data.AutoSave) == "boolean" then
					self.Meta.AutoSave = data.AutoSave
				end
				if type(data.AutoLoad) == "boolean" then
					self.Meta.AutoLoad = data.AutoLoad
				end
				for _, k in {"AutoBind", "Onboarded", "TimeSwitch"} do
					if type(data[k]) == "boolean" then
						self.Meta[k] = data[k]
					end
				end
				if type(data.Binds) == "table" then
					self.Meta.Binds = {}
					for k, v in data.Binds do
						if type(v) == "string" then
							self.Meta.Binds[tostring(k)] = v
						end
					end
				end
				for _, k in {"DayTheme", "NightTheme"} do
					if type(data[k]) == "string" then
						self.Meta[k] = data[k]
					end
				end
				for _, k in {"NightFrom", "NightTo"} do
					if type(data[k]) == "number" then
						self.Meta[k] = math.clamp(math.floor(data[k]), 0, 23)
					end
				end
			end
		end
	end)
end

		return false

		self:Notify("配置", "没有这份备份", 2, "warning")
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(path))
	end)

		self:Notify("配置", "备份文件损坏", 2, "alert")
		return false

	self:Deserialize(data)

	self:Notify("配置", "已回滚到备份 " .. index, 1.8, "info")
	return true
end

function Library:SaveConfig(name, silent)
	if not self.CanFile then
		if not silent then
			self:Notify("配置", "当前环境不支持文件操作", 2.5, "warning")
		end
		return false
	end
	name = sanitize(name)
	local ok, err = pcall(function()
		if not isfolder(self.Folder) then
			makefolder(self.Folder)
		end
		local path = self.Folder .. "/" .. name .. ".json"
		if isfile(path) then
			local now = os.time()
			self.BackupAt = self.BackupAt or {}
			if not silent or now - (self.BackupAt[name] or 0) >= 180 then
				self.BackupAt[name] = now
				for i = 3, 2, -1 do
					local from = self:BackupPath(name, i - 1)
					if isfile(from) then
						writefile(self:BackupPath(name, i), readfile(from))
					end
				end
				writefile(self:BackupPath(name, 1), readfile(path))
			end
		end
		writefile(path, HttpService:JSONEncode(self:Serialize()))
	end)
	if ok then
		self.Meta.Config = name
		self:SaveMeta()
		if not silent then
			self:Notify("配置", "已保存：" .. name, 1.8, "info")
		end
	else
		warn("[ClickGui] " .. tostring(err))
		if not silent then
			self:Notify("配置", "保存失败", 2.5, "alert")
		end
	end
	return ok
end

function Library:LoadConfig(name, silent, part)
	if not self.CanFile then
		if not silent then
			self:Notify("配置", "当前环境不支持文件操作", 2.5, "warning")
		end
		return false
	end
	name = sanitize(name)
	local path = self.Folder .. "/" .. name .. ".json"
	local okExists, exists = pcall(isfile, path)
	if not okExists or not exists then
		if not silent then
			self:Notify("配置", "找不到配置：" .. name, 2.5, "warning")
		end
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(path))
	end)
	if not ok or type(data) ~= "table" then
		if not silent then
			self:Notify("配置", "配置文件损坏", 2.5, "alert")
		end
		return false
	end
	local payload = data
	if part == "外观" then
		payload = {Theme = data.Theme, Categories = data.Categories, CustomCats = data.CustomCats}
	elseif part == "功能" then
		payload = {Modules = data.Modules}
	end
	self:Deserialize(payload)
	if not part then
		self.Meta.Config = name
		self:SaveMeta()
	else
		self:QueueSave()
	end
	if not silent then
		self:Notify("配置", "已加载" .. (part and (part .. "：") or "：") .. name, 1.8, "info")
	end
	return true
end

function Library:DeleteConfig(name)

	name = sanitize(name)

		end
	end)
	if ok then

	end

end

	self:RefreshFilter()

function Library:RefreshFilter()
	local content = self.Search.Text:lower()
	for _, module in self.Modules do
		local okSearch = content == "" or module.Name:lower():find(content, 1, true) ~= nil or (module.Display ~= nil and module.Display:lower():find(content, 1, true) ~= nil)
		if not okSearch and content:match("^%a+$") then
			okSearch = string.find(pinyinOf(module.Name), content, 1, true) ~= nil or (module.Display ~= nil and string.find(pinyinOf(module.Display), content, 1, true) ~= nil)
		end
		local okFilter = self.Filter == "全部" or (self.Filter == "已启用" and module.Enabled) or (self.Filter == "未启用" and not module.Enabled) or (self.Filter == "最近" and table.find(self.Recent, module.Key) ~= nil) or (self.Filter == "收藏" and module.Pinned == true)
		module.Container.Visible = okSearch and okFilter and not module.Hidden
	end
end

function Library:SetVisible(state)
	if self.Visible == state then return end
	self.Visible = state
	self:UpdateGrip()
	local dim = 1 - self.Theme.Dim / 100
	if state then
		self.Root.Visible = true
		self:Onboard()
		if animEnabled then
			self.Root.BackgroundTransparency = 1
			self.OpenScale.Scale = 0.94
			tween(self.Root, {BackgroundTransparency = dim}, 0.28)
			tween(self.OpenScale, {Scale = 1}, 0.28)
			if self.Theme.Stagger and not self.Theme.Perf then
				local i = 0
				for _, cat in self.Categories do
					if not cat.Floating then
						cat:PlayEntrance(i * 0.045)
						i += 1
					end
				end
			end
		else
			self.Root.BackgroundTransparency = dim
			self.OpenScale.Scale = 1
		end
		tween(self.Blur, {Size = self:BlurSize()}, 0.28)
	else
		tween(self.Root, {BackgroundTransparency = 1}, 0.2)
		tween(self.Blur, {Size = 0}, 0.2)
		task.delay(0.2, function()
			if not self.Visible then
				self.Root.Visible = false
			end
		end)
	end
end

function Library:Lock(state)
	self.Scroller.ScrollingEnabled = not state
	for _, category in self.Categories do
		category.Body.ScrollingEnabled = not state
	end
end

function Library:KillToast(rec)
	if not rec.Alive then return end
	rec.Alive = false
	local i = table.find(self.ToastList, rec)
	if i then
		table.remove(self.ToastList, i)
	end
	if self.ToastMap[rec.Key] == rec then
		self.ToastMap[rec.Key] = nil
	end
	rec.Frame:Destroy()
end

function Library:Notify(title, content, duration, kind)
	if not self.Theme.Notifications then return end
	duration = duration or 1.6
	local icon = kind == "warning" and Assets.Warning or kind == "alert" and Assets.Alert or Assets.Info
	local existing = self.ToastMap[title]
	if existing and existing.Alive then
		existing.Token += 1
		local token = existing.Token
		existing.Content.Text = content
		existing.Icon.Image = icon
		existing.Progress.Size = UDim2.new(1, -12, 0, 2)
		tween(existing.Progress, {Size = UDim2.fromOffset(0, 2)}, duration)
		task.delay(duration, function()
			if existing.Alive and existing.Token == token then
				self:KillToast(existing)
			end
		end)
		return
	end
	while #self.ToastList >= 3 do
		self:KillToast(self.ToastList[1])
	end
	local p = self:Palette(1)
	self.ToastOrder += 1
	local frame = create("Frame", {
		Size = UDim2.fromOffset(230, 54),
		BorderSizePixel = 0,
		ClipsDescendants = true,
		LayoutOrder = self.ToastOrder
	}, self.Toasts)
	local style = makeGlass(frame, self.Theme.Corner)
	style(self, p, p.Bg, self.Theme.Corner)
	local iconLabel = create("ImageLabel", {
		Size = UDim2.fromOffset(26, 26),
		Position = UDim2.fromOffset(12, 13),
		BackgroundTransparency = 1,
		Image = icon,
		ImageColor3 = p.Text
	}, frame)
	create("TextLabel", {
		Size = UDim2.new(1, -60, 0, 20),
		Position = UDim2.fromOffset(48, 8),
		BackgroundTransparency = 1,
		Text = title,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextColor3 = p.Text,
		TextSize = 16,
		Font = T.FontBold
	}, frame)
	local contentLabel = create("TextLabel", {
		Size = UDim2.new(1, -60, 0, 18),
		Position = UDim2.fromOffset(48, 28),
		BackgroundTransparency = 1,
		Text = content,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextColor3 = p.Sub,
		TextSize = 14,
		Font = T.Font,
		TextTruncate = Enum.TextTruncate.AtEnd
	}, frame)

		Size = UDim2.new(1, -12, 0, 2),
		Position = UDim2.new(0, 6, 1, -4),

	self.ToastMap[title] = rec
	task.delay(duration, function()

			self:KillToast(rec)
		end
	end)
end

function Library:HudColorize(hue)
	local th = self.Theme
	local p = self:Palette(1)
	local list = self.Hud.Sorted
	if not list then return end
	local mode = th.HudColor
	hue = hue or self.Hue
	local custom = W
	if mode == "自定义" then
		local ok, parsed = pcall(Color3.fromHex, th.HudHex)
		if ok then
			custom = parsed
		end
	end
	for i, rec in list do
		local c
		if mode == "彩虹" then
			c = Color3.fromHSV((hue - (i - 1) * 0.045) % 1, 0.65, 1)
		elseif mode == "自定义" then
			c = custom:Lerp(B, math.clamp((i - 1) * 0.06, 0, 0.45))
		else
			c = p.Accent:Lerp(p.Sub, math.clamp((i - 1) * 0.12, 0, 0.75))
		end
		rec.Label.TextColor3 = c
		rec.Bar.BackgroundColor3 = c
	end
end

function Library:RefreshHud()
	local hud = self.Hud
	local th = self.Theme
	local p = self:Palette(1)
	local pos = th.HudPos
	local right = pos == "右上" or pos == "右下"
	local bottom = pos == "右下" or pos == "左下"
	local list = {}
	for module, rec in hud.Entries do
		local extra = extraOf(module)
		local shown = module.Display or module.Name
		local plain = shown .. (extra or "")
		rec.Label.Text = shown .. (extra and (' <font color="#' .. p.Sub:ToHex() .. '">' .. extra .. "</font>") or "")
		rec.Length = utf8.len(plain) or #plain
		rec.Label.TextSize = th.FontSize
		textShadow(rec.Label, th)
		rec.Entry.BackgroundColor3 = p.Bg
		rec.Entry.BackgroundTransparency = th.HudBg and math.clamp(th.Transparency + 0.3, 0, 1) or 1
		rec.Bar.Visible = th.HudBar
		rec.Bar.LayoutOrder = right and 2 or 0
		table.insert(list, rec)
	end
	table.sort(list, function(a, c)
		if th.HudSort == "字数" and a.Length ~= c.Length then
			return a.Length > c.Length
		end
		return a.Module.Name < c.Module.Name
	end)
	for i, rec in list do
		rec.Holder.LayoutOrder = bottom and -i or i
	end
	hud.Sorted = list
	self:HudColorize()
end

function Library:HudRemove(module, immediate)
	local hud = self.Hud

		rec.Holder:Destroy()

		tween(rec.Label, {TextTransparency = 1}, 0.2)
		tween(rec.Bar, {BackgroundTransparency = 1}, 0.2)
		task.delay(0.25, function()
			rec.Holder:Destroy()
		end)
	end
	self:RefreshHud()
end

function Library:UpdateFps()
	local th = self.Theme
	local need = th.FpsShow or (th.AutoPerf and not th.Perf)
	if self.FpsLabel then
		self.FpsLabel.Visible = th.FpsShow
	end
	if need and not self.FpsConn then
		self.FpsAcc, self.FpsFrames, self.FpsLow = 0, 0, 0
		self.FpsConn = RunService.Heartbeat:Connect(function(dt)
			self.FpsAcc += dt
			self.FpsFrames += 1
			if self.FpsAcc >= 0.5 then
				local fps = math.floor(self.FpsFrames / self.FpsAcc + 0.5)
				self.Fps = fps
				self.FpsAcc, self.FpsFrames = 0, 0
				if self.FpsLabel then
					self.FpsLabel.Text = fps .. " FPS"
				end
				if self.Theme.AutoPerf and not self.Theme.Perf then
					if fps < 24 then
						self.FpsLow += 1
					else
						self.FpsLow = 0
					end
					if self.FpsLow >= 12 then
						self.FpsLow = 0
						self.Theme.Perf = true
						if self.Controls.Perf then
							self.Controls.Perf:SetValue(true, true)
						end
						self:ApplyTheme()
						self:QueueSave()
						self:Notify("性能", "帧率过低，已自动开启性能模式", 3, "warning")
					end
				end
			end
		end)
		self.Maid:Give(self.FpsConn)
	elseif not need and self.FpsConn then
		self.FpsConn:Disconnect()
		self.FpsConn = nil
	end
end

function Library:CheckTimeTheme()
	local m = self.Meta
	if not m.TimeSwitch then return end
	local ok, hour = pcall(function()
		return tonumber(os.date("%H"))
	end)
	if not ok or not hour then return end
	local night
	if m.NightFrom == m.NightTo then
		night = false
	elseif m.NightFrom < m.NightTo then
		night = hour >= m.NightFrom and hour < m.NightTo
	else
		night = hour >= m.NightFrom or hour < m.NightTo
	end
	local target = night and m.NightTheme or m.DayTheme
	if target == "" or target == "（无）" or target == self.TimeApplied then return end
	if self:LoadTheme(target) then
		self.TimeApplied = target
	end
end

function Library:StartTimeLoop()
	if self.TimeLoop then return end
	self.TimeLoop = true
	local function step()
		if self.Destroyed then return end
		self:CheckTimeTheme()
		task.delay(60, step)
	end
	step()
end

function Library:Onboard()
	if self.Meta.Onboarded or not self.IsReady then return end
	self.Meta.Onboarded = true
	self:SaveMeta()
	local tips = {
		"点功能开关，点右侧的 ⋮ 展开它的设置",
		"长按功能可以放置快捷按钮",
		"设置里可以搜索、撤销和预览外观"
	}
	for i, tip in tips do
		task.delay(0.5 + (i - 1) * 3, function()
			if not self.Destroyed and self.Visible then
				self:Notify("提示 " .. i, tip, 2.8, "info")
			end
		end)
	end
end

function Library:Ready()
	if self.IsReady then
		return self
	end
	if self.Meta.AutoLoad then
		local name = self.Meta.Config
		if self.Meta.AutoBind then
			local bound = self.Meta.Binds[tostring(game.PlaceId)]
			if bound then
				name = bound
			end
		end
		self:LoadConfig(name, true)
	end
	self.IsReady = true
	self:RefreshModList()
	self:StartTimeLoop()
	return self
end

function Library:BuildSettings(category)
	local theme = self.Theme
	local function P()
		return self:Palette(1)
	end
	local b = Builder(self, category.Body, P, nil, nil, nil)
	local controls = self.Controls

	local function set(key, convert)
		return function(v)
			if convert then
				theme[key] = convert(v)
			else
				theme[key] = v
			end
			self:QueueTheme()
			self:QueueSave()
		end
	end

	local sectionList = {}
	self.SettingSections = sectionList
	local sectionOrder = 20
	local searchQuery = ""

	local function lastItem(box)
		local best
		for _, c in box:GetChildren() do
			if c:IsA("GuiObject") and (not best or c.LayoutOrder >= best.LayoutOrder) then
				best = c
			end
		end
		return best
	end

	local function paintSection(entry)
		entry.Arrow.Text = entry.Open and "▾" or "▸"
	end

	local function section(title, build, open)
		sectionOrder += 2
		local entry = {Title = title, Built = false, Open = false, Items = {}}
		local head = create("TextButton", {
			Size = UDim2.new(1, 0, 0, 32),
			BackgroundTransparency = 1,
			Text = "",
			AutoButtonColor = false,
			LayoutOrder = sectionOrder
		}, category.Body)
		entry.Arrow = create("TextLabel", {
			Size = UDim2.fromOffset(20, 32),
			Position = UDim2.fromOffset(10, 0),
			BackgroundTransparency = 1,
			Text = "▸",
			TextSize = 14,
			Font = T.FontBold
		}, head)
		local label = create("TextLabel", {
			Size = UDim2.new(1, -40, 1, 0),
			Position = UDim2.fromOffset(30, 0),
			BackgroundTransparency = 1,
			Text = title,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextSize = 14,
			Font = T.FontBold
		}, head)
		local line = create("Frame", {
			Size = UDim2.new(1, -28, 0, 1),
			Position = UDim2.new(0, 14, 1, -2),
			BorderSizePixel = 0
		}, head)
		local box = create("Frame", {
			Size = UDim2.new(1, 0, 0, 0),
			AutomaticSize = Enum.AutomaticSize.Y,
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
			Visible = false,
			LayoutOrder = sectionOrder + 1
		}, category.Body)
		create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder}, box)
		entry.Head = head
		entry.Box = box
		self:Reg(function()
			local p = P()
			label.TextColor3 = p.Text
			entry.Arrow.TextColor3 = p.Sub
			line.BackgroundColor3 = p.Off
		end)
		function entry.Build()
			if entry.Built then return end
			entry.Built = true
			local sb = Builder(self, box, P, nil, nil, nil)
			local proxy = setmetatable({}, {__index = function(_, k)
				local f = sb[k]
				if type(f) ~= "function" then
					return f
				end
				return function(_, arg)
					local r = f(sb, arg)
					if k ~= "CreateDivider" then
						local item = lastItem(box)
						if item then
							table.insert(entry.Items, {Item = item, Name = string.lower(type(arg) == "table" and tostring(arg.Name) or tostring(arg))})
						end
					end
					return r
				end
			end})
			local ok, err = pcall(build, proxy)
			if not ok then
				warn("[ClickGui] " .. title .. ": " .. tostring(err))
			end
		end
		function entry.SetOpen(state)
			entry.Open = state
			if state then
				entry.Build()
			end
			box.Visible = state
			paintSection(entry)
		end
		head.MouseButton1Click:Connect(function()
			if searchQuery == "" then
				entry.SetOpen(not entry.Open)
			end
		end)
		table.insert(sectionList, entry)
		if open then
			entry.SetOpen(true)
		end
		return entry
	end

	local function applyFilter()
		local q = string.lower(searchQuery)
		for _, e in sectionList do
			if q == "" then
				e.Head.Visible = true
				e.Box.Visible = e.Open
				for _, it in e.Items do
					it.Item.Visible = true
				end
			else
				e.Build()
				local titleHit = string.find(string.lower(e.Title), q, 1, true) ~= nil
				if not titleHit and q:match("^%a+$") then
					titleHit = string.find(pinyinOf(e.Title), q, 1, true) ~= nil
				end
				local any = titleHit
				for _, it in e.Items do
					local hit = titleHit or string.find(it.Name, q, 1, true) ~= nil
				if not hit and q:match("^%a+$") then
					hit = string.find(pinyinOf(it.Name), q, 1, true) ~= nil
				end
					it.Item.Visible = hit
					any = any or hit
				end
				e.Head.Visible = any
				e.Box.Visible = any
			end
			paintSection(e)
		end
	end

	}
	category.Order = category.Index
	category.PIndex = settings.PaletteIndex or category.Index
	local function curWidth()
		return settings.Width or category.WidthOverride or window.Theme.PanelW
	end

	local function P()
		local base = window:Palette(category.PIndex)
		local hex = category.ColorHex
		if not hex then
			return base
		end
		if category.PalBase == base and category.PalHex == hex and category.PalMix == window.Theme.TintMix then
			return category.PalCache
		end
		local ok, c = pcall(Color3.fromHex, hex)
		if not ok then
			return base
		end
		local mix = math.clamp(window.Theme.TintMix / 100, 0.35, 1)
		local p = {}
		for k, v in base do
			p[k] = v
		end
		p.RowOn = base.RowOn:Lerp(c, mix)
		p.Fill = base.Fill:Lerp(c, mix)
		p.Rim1 = base.Rim1:Lerp(c, math.clamp(mix * 1.6, 0, 1))
		p.OnText = luminance(p.RowOn) > 0.55 and Color3.fromRGB(10, 10, 12) or Color3.fromRGB(244, 244, 248)
		p.Accent = p.Fill
		p.Tinted = true
		p.TintColor = c
		category.PalBase, category.PalHex, category.PalMix, category.PalCache = base, hex, window.Theme.TintMix, p
		return p
	end

	category.GetPal = P

		if ps == "实心" then
			panel.BackgroundTransparency = 0
		elseif ps == "轻描边" then

			panelStroke.Thickness = math.max(th.StrokeW, 1.5)

		headGrad.Rotation = th.GradAngle
		headGrad.Enabled = not perf
		header.BackgroundColor3 = perf and p.Head or W
		header.BackgroundTransparency = math.clamp(th.Transparency - 0.05, 0, 1)
		title.TextColor3 = p.HeadText
		title.TextSize = th.FontSize + 2
		textShadow(title, th)
		body.ScrollBarThickness = th.BarThick
		layout.Padding = UDim.new(0, th.RowGap)
		arrow.ImageColor3 = p.HeadSub
		if icon then
			icon.ImageColor3 = p.HeadText
		end
		bottomLine.BackgroundColor3 = p.Tinted and p.Accent or p.Rim2
		for _, eb in editButtons do

			eb.TextColor3 = p.OnText

	local function resize()
		local s = window.Scale.Scale
		local maxHeight = math.max(window.Root.AbsoluteSize.Y / s - 130, 120)
		local content = layout.AbsoluteContentSize.Y / s
		local target = category.Collapsed and 0 or math.min(content, maxHeight)
		local goal = UDim2.fromOffset(curWidth(), 36 + target)
		if animEnabled and window.IsReady then
			tween(holder, {Size = goal}, 0.18)
		else
			holder.Size = goal
		end
	end
	layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(resize)
	self.Root:GetPropertyChangedSignal("AbsoluteSize"):Connect(resize)
	self.Scale:GetPropertyChangedSignal("Scale"):Connect(resize)
	task.defer(resize)
	self:Reg(function()
		resize()
	end)

	category.Resize = resize
	category.Header = header

	function category:SetFree(state)
		if settings.Floating then return end
		if state then
			if not category.FreePos then
				local s = window.Scale.Scale
				local ap = holder.AbsolutePosition
				local rp = window.Root.AbsolutePosition
				category.FreePos = {X = (ap.X - rp.X) / s, Y = (ap.Y - rp.Y) / s}
			end
			category.Free = true
			holder.Parent = window.Root
			holder.Position = UDim2.fromOffset(category.FreePos.X, category.FreePos.Y)
		else
			category.Free = false
			holder.Parent = window.Scroller
			holder.Position = UDim2.fromOffset(0, 0)
			holder.ZIndex = 1
		end
	end

	function category:ApplyStyle()
		category:ApplyIcon()
		holder.LayoutOrder = category.Order
		if not settings.Floating then
			holder.Visible = not category.Hidden
		end
		resize()
	end

		panel.Position = UDim2.fromOffset(0, 30)
		shadow.Position = UDim2.fromOffset(-48, -1)

			tween(panel, {Position = UDim2.fromOffset(0, 0)}, 0.32)
			tween(shadow, {Position = UDim2.fromOffset(-48, -31)}, 0.32)
		end)
	end

	function category:SetCollapsed(state)
		self.Collapsed = state
		tween(arrow, {Rotation = state and 180 or 0}, 0.18)
		resize()
	end

	local function collapse()
		category:SetCollapsed(not category.Collapsed)
		window:QueueSave()
	end

		self:MakeDraggable(header, holder, {
			OnTap = collapse,
			GetScale = function()
				return window.Scale.Scale
			end
		})

		self:MakeDraggable(header, holder, {
			OnTap = collapse,
			GetScale = function()
				return window.Scale.Scale
			end,
			CanMove = function()
				return category.Free == true
			end,
			OnDown = function()
				if category.Free then
					window:RaiseCat(category)
				end
			end,
			OnMoved = function()
				if category.Free then
					window:ClampCat(category)
					window:QueueSave()
				end
			end
		})
	end

			local rainbow = th.AccentRainbow and not perf and not custom
			local filled = on and (style == "渐变填充" or style == "纯色填充")
			local glossy = filled and style == "渐变填充" and not perf
			local accent = custom or p.RowOn
			local spec = on and not custom and not rainbow and th.EnabledColor ~= "主题色"
			if on and rainbow then
				accent = Color3.fromHSV(window.Hue, 0.55, 1)
			elseif spec then
				accent = window:SpectrumColor(row, module)
			end
			rowGrad.Color = sheen(accent, th.Gloss / 100)
			rowGrad.Enabled = glossy
			row.Size = UDim2.new(1, 0, 0, th.RowH)
			expand.Size = UDim2.fromOffset(34, th.RowH)
			local bg, bgT
			if filled then
				bg = glossy and W or accent
				bgT = 0
			else
				bg = p.Row
				bgT = math.clamp(th.Transparency + (on and 0.15 or 0.3), 0, 1)
			end
			local textColor
			if filled then
				textColor = rainbow and DARK.OnText or (spec and (luminance(accent) > 0.55 and DARK.OnText or LIGHT.OnText)) or (custom and (luminance(custom) > 0.55 and DARK.OnText or LIGHT.OnText) or p.OnText)
			else
				textColor = on and p.Text or p.Sub
			end
			local hasIcon = module.IconId ~= nil
			rowIcon.Visible = hasIcon
			if hasIcon then
				rowIcon.Image = "rbxassetid://" .. module.IconId
				rowIcon.ImageColor3 = textColor
			end
			label.Position = UDim2.new(0, hasIcon and 34 or 14, 0.5, 0)
			label.Size = UDim2.new(1, hasIcon and -70 or -50, 1, 0)
			apply(row, {BackgroundColor3 = bg, BackgroundTransparency = bgT}, animate)
			apply(label, {TextColor3 = textColor}, animate)
			apply(dots, {ImageColor3 = textColor}, animate)
			label.Font = on and T.FontBold or T.Font
			label.TextSize = th.FontSize
			textShadow(label, th)
			local ac = custom or ((on and (rainbow or spec)) and accent or p.Accent)
			leftBar.Visible = on and style == "左侧光条"
			leftBar.BackgroundColor3 = ac
			rowStroke.Enabled = on and style == "发光描边"
			rowStroke.Color = ac
			shine.Visible = on and th.Shimmer and not perf
			settingsFrame.BackgroundColor3 = p.Set
			settingsFrame.BackgroundTransparency = 0.25
		end
		window:Reg(refresh)
		module.Refresh = refresh
		row:GetPropertyChangedSignal("AbsolutePosition"):Connect(function()
			if module.Enabled and window.Theme.EnabledColor ~= "主题色" and not module.RecolorPending then
				module.RecolorPending = true
				task.delay(0.12, function()
					module.RecolorPending = false
					if module.Enabled and not window.Destroyed then
						refresh(false)
					end
				end)
			end
		end)

		function module:ApplyCustom()
			label.Text = self.Display or self.Name
			container.LayoutOrder = self.Pinned and (self.Index - 1000) or self.Index
			refresh(false)
		end

		function module:Toggle(state, silent)
			if state == nil then
				state = not self.Enabled
			end
			if state and not silent and not self.Enabled then
				for _, other in window.Conflicts[self.Key] or {} do
					local m2 = window.ModulesByKey[other]
					if m2 and m2.Enabled then
						m2:Toggle(false, true)
						window:Notify("冲突", m2.Name .. " 与 " .. self.Name .. " 冲突，已自动关闭", 2.4, "warning")
					end
				end
			end
			self.Enabled = state
			refresh(true)
			local quick = window.Quicks[self.Key]
			if quick then
				quick.Refresh(true)
			end
			window:HudUpdate(self)
			if window.Filter ~= "全部" then
				window:RefreshFilter()
			end
			local px = self.PressX
			self.PressX = nil
			if not silent then
				local p = P()
				window:PushRecent(module.Key)
				window:Click()
				window:Wave(row, px, state and (module.ColorC or p.Fill) or p.Text)
				window:Notify(modSettings.Name, state and "已开启" or "已关闭", 1.2, state and "info" or "warning")
				window:QueueSave()
			end
			window:Safe(modSettings.Name, callback, state)
		end

		window:MakeDraggable(row, row, {
			NoMove = true,
			HoldTime = 0.7,
			OnDown = function(input)
				local p = P()
				module.PressX = (input.Position.X - row.AbsolutePosition.X) / window.Scale.Scale
				window:Ripple(row, input, module.Enabled and p.OnText or p.Text)
				if window.Theme.Ripple and not window.Theme.Perf then
					tween(labelScale, {Scale = 0.95}, 0.08)
				end
			end,
			OnUp = function()
				tween(labelScale, {Scale = 1}, 0.16)
			end,
			OnTap = function()
				module:Toggle()
			end,
			OnHold = function()
				window:BeginPlace(module)
			end
		})

		local settingsOpen = false
		local function settingsHeight()
			return settingsFrame.AbsoluteSize.Y / window.Scale.Scale
		end
		function module:SetSettingsOpen(state)
			settingsOpen = state
			tween(dots, {Rotation = state and 90 or 0}, 0.18)
			apply(settingsClip, {Size = UDim2.new(1, 0, 0, state and settingsHeight() or 0)}, true)
			task.delay(0.24, function()
				if settingsOpen then
					settingsClip.Size = UDim2.new(1, 0, 0, settingsHeight())
				end
			end)
		end
		settingsFrame:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
			if settingsOpen then
				settingsClip.Size = UDim2.new(1, 0, 0, settingsHeight())
			end
		end)
		expand.MouseButton1Click:Connect(function()
			module:SetSettingsOpen(not settingsOpen)
		end)

		local b = Builder(window, settingsFrame, P, module.Options, function()
			expand.Visible = true
		end, function()
			if module.Enabled then
				window:RefreshHud()
			end
			window:QueueSave()
		end)
		for _, name in {"CreateToggle", "CreateSlider", "CreateDropdown", "CreateTextBox", "CreateButton", "CreateColorPicker", "CreateDivider"} do
			module[name] = function(_, s)
				return b[name](b, s)
			end
		end

		module.Object = row
		table.insert(category.Modules, module)
		table.insert(window.Modules, module)
		window.ModulesByKey[module.Key] = module
		if window.IsReady then
			window:RefreshModList()
		end

		if modSettings.Default then
			module:Toggle(true, true)
		end

		return module
	end

	table.insert(self.Categories, category)
	self:RefreshCatList()
	return category
end

local UI = {
	Version = Library.Version,
	Icons = Assets,
	Window = nil
}

function UI:CreateWindow(options)
	if self ~= UI then
		options = self
	end
	options = options or {}
	local current = UI.Window
	if current and not current.Destroyed then
		pcall(function()
			current:Destroy()
		end)
	end
	UI.Window = nil
	if options.Folder then
		Library.Folder = options.Folder
	end
	local window = Library.new(options)
	UI.Window = window
	task.defer(function()
		if not window.Destroyed and not window.IsReady then
			window:Ready()
		end
	end)
	return window
end

UI.new = UI.CreateWindow

function UI:GetWindow()
	local window = UI.Window
	if window and not window.Destroyed then
		return window
	end
	return nil
end

function UI:SetVisible(state)
	local window = UI:GetWindow()
	if window then
		window:SetVisible(state)
	end
end

function UI:Toggle()
	local window = UI:GetWindow()
	if window then
		window:SetVisible(not window.Visible)
	end
end

function UI:Destroy()
	local window = UI.Window
	UI.Window = nil
	if window and not window.Destroyed then
		pcall(function()
			window:Destroy()
		end)
	end
	if env.__CLICKGUI_LIB == UI then
		env.__CLICKGUI_LIB = nil
	end
end

env.__CLICKGUI_LIB = UI

return UI
